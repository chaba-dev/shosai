/// Follow-up checks on a real device (uncommitted helper, 2026-11-02 run
/// against claude/project-thread-r96hxq-verify-all 0c19676): PDF tap-turn
/// latency and its place line, a steady stage under a curl, the Aa panel's
/// scrim, the PDF place line on a tablet, and EPUB page numbers through the
/// book.
///
/// Prints `SHOT <label>` and holds still so a host watcher can take a real
/// device screenshot; `QP <key>: <value>` lines carry measurements; `FRAMES
/// <dir>` names in-app frame captures (with `<name>-NNN-<ms>.png` names,
/// `ms` from the tap or drag start) copied off the device afterwards.
const _device = String.fromEnvironment('QP_DEVICE', defaultValue: 'device');
const _localeTag = String.fromEnvironment('QP_LOCALE', defaultValue: 'en');
const _books = String.fromEnvironment('QP_BOOKS', defaultValue: 'all');
final _captureKey = GlobalKey();
late Directory _frames;
int _step = 0;
// EPUB taps go just below the text, clear of links (the contents spread).
var _marginTaps = false;

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  binding.framePolicy = LiveTestWidgetsFlutterBindingFramePolicy.fullyLive;

  testWidgets('follow-ups', timeout: const Timeout(Duration(minutes: 60)), (
    tester,
  ) async {
    final books = Directory.systemTemp.createTempSync('qp-books-');
    for (final name in ['pride.epub', 'attention.pdf']) {
      final data = await rootBundle.load('qp-books/$name');
      File(
        '${books.path}/$name',
      ).writeAsBytesSync(data.buffer.asUint8List(), flush: true);
    }
    _frames = Directory.systemTemp.createTempSync('qp-frames-');
    _out('FRAMES ${_frames.path}');
    final data = Directory.systemTemp.createTempSync('qp-data-');
    _note('data-dir', data.path);
    final owner = await applicationCoreOwner(
      applicationSupportDirectory: () async => data,
    );
    final bridge = owner.bridge();
    final imported = await bridge.importPaths(
      pathKeys: [for (final f in books.listSync()) f.path],
      managed: false,
      cancellationId: bridge.createCancellation(),
    );
    _note('imported', imported.imported.toInt());

    await tester.pumpWidget(
      RepaintBoundary(
        key: _captureKey,
        child: ShosaiApp(
          productBridgeFactory: owner.bridge,
          locale: Locale(_localeTag),
        ),
      ),
    );
    await _settle(tester, seconds: 3);
    _note(
      'view',
      'logical ${tester.view.physicalSize.width / tester.view.devicePixelRatio}'
          'x${tester.view.physicalSize.height / tester.view.devicePixelRatio}'
          ' dpr ${tester.view.devicePixelRatio}'
          ' refresh ${tester.view.display.refreshRate} locale $_localeTag',
    );
    await _shot(tester, 'library-start');

    if (_books == 'round3') {
      await _pdfShadow(tester);
      await _f2r3(tester);
    }
    if (_books == 'all' || _books == 'pdf') await _pdf(tester, binding);
    if (_books == 'all' || _books == 'epub' || _books == 'numbers') {
      await _epub(tester);
    }

    await _try('end-library', () async {
      if (_stage.evaluate().isNotEmpty) await _toLibrary(tester);
      _note('end-in-library', _stage.evaluate().isEmpty);
      await _shot(tester, 'library-end');
    });
    final error = tester.takeException();
    _note('final-exception', error);
    _out('DONE');
  });
}

Future<void> _pdf(
  WidgetTester tester,
  IntegrationTestWidgetsFlutterBinding binding,
) async {
  await _try('pdf-open', () async {
    if (_stage.evaluate().isNotEmpty) await _toLibrary(tester);
    await _open(tester, 'ttention');
    await _dismissHint(tester);
    await _hideChrome(tester);
    _note('pdf-spread', _spread(tester));
    // Off the first page, so both turn directions are open.
    await _tapAt(tester, .85, settleMs: 1200);
    await _tapAt(tester, .85, settleMs: 1200);
    await _hideChrome(tester);
    await _shot(tester, 'pdf-quiet-place-line');
  });

  // 1. Tap turns: time from the finger lifting to the curl's first frame and
  // to its leaf moving, and when the folio widget changes.
  await _try('pdf-tap-latency', () async {
    await _hideChrome(tester);
    final firsts = <int>[];
    final moves = <int>[];
    for (var i = 0; i < 10; i++) {
      final f = i.isEven ? .85 : .15;
      final r = await _timedTap(tester, f, 'pdf-tap-$i', capture: false);
      _note('pdf-tap-$i ${f == .85 ? 'forward' : 'back'}', r.summary);
      if (r.firstCurlMs != null) firsts.add(r.firstCurlMs!);
      if (r.firstMoveMs != null) moves.add(r.firstMoveMs!);
      await _settle(tester, ms: 500);
    }
    _note('pdf-tap-first-curl-ms', _stats(firsts));
    _note('pdf-tap-first-leaf-move-ms', _stats(moves));
  });

  // 1 and 2, recorded: every frame of a forward and a back tap turn.
  await _try('pdf-tap-recorded', () async {
    await _hideChrome(tester);
    for (final (name, f) in [('forward', .85), ('back', .15)]) {
      final r = await _timedTap(tester, f, 'pdf-rec-$name', capture: true);
      _note('pdf-rec-$name', r.summary);
      await _settle(tester, ms: 600);
    }
  });

  // 2. A held forward drag: the stage and shadow around the page.
  await _try('pdf-held', () async {
    await _hideChrome(tester);
    await _heldDrag(tester, 'pdf');
  });

  // 3. The Aa panel's scrim.
  await _try('pdf-aa', () async => _aaPanel(tester, 'pdf'));

  await _try('pdf-timing', () async {
    await _hideChrome(tester);
    final timings = <ui.FrameTiming>[];
    void collect(List<ui.FrameTiming> batch) => timings.addAll(batch);
    binding.addTimingsCallback(collect);
    for (var i = 0; i < 8; i++) {
      await _tapAt(tester, i.isEven ? .85 : .15, settleMs: 700);
    }
    await _settle(tester, seconds: 2);
    binding.removeTimingsCallback(collect);
    _note('pdf-frames', _timingSummary(timings, tester));
  });
}

Future<void> _epub(WidgetTester tester) async {
  _marginTaps = true;
  await _try('epub-open', () async {
    if (_stage.evaluate().isNotEmpty) await _toLibrary(tester);
    await _open(tester, 'Pride');
    await _dismissHint(tester);
    await _hideChrome(tester);
    await _waitNumbered(tester);
    _note('epub-spread', _spread(tester));
  });

  if (_device == 'ipad' && !_spread(tester)) {
    await _try('epub-two-pages', () async {
      await _openAa(tester);
      await _tapKey(tester, 'reader-typography-two-pages');
      await _dismissPanel(tester);
      await _hideChrome(tester);
      await _settle(tester, seconds: 2);
      _note('epub-spread-after', _spread(tester));
    });
  }

  // 5a. The opening pages and their numbers.
  if (_books != 'numbers') await _try('epub-opening', () async {
    await _hideChrome(tester);
    // Back to the cover.
    for (var i = 0; i < 6; i++) {
      final before = _location(tester);
      await _tapAt(tester, .15, settleMs: 900);
      if (_location(tester) == before) break;
    }
    await _waitNumbered(tester);
    _note('epub-open-0', _location(tester));
    await _shot(tester, 'epub-cover');
    for (var i = 1; i <= 4; i++) {
      await _tapAt(tester, .85, settleMs: 1000);
      await _waitNumbered(tester);
      _note('epub-open-$i', _location(tester));
      if (i == 1) await _shot(tester, 'epub-after-cover');
    }
  });

  // 3. The Aa panel's scrim over an EPUB.
  if (_books != 'numbers') await _try('epub-aa', () async => _aaPanel(tester, 'epub'));

  // 2. A held drag and a recorded tap turn on an EPUB page or spread.
  if (_books != 'numbers') await _try('epub-held', () async {
    await _hideChrome(tester);
    await _heldDrag(tester, 'epub');
    final r = await _timedTap(tester, .85, 'epub-rec-forward', capture: true);
    _note('epub-rec-forward', r.summary);
    await _settle(tester, ms: 600);
  });

  // 5b. A late chapter from Contents, then the same page reached by paging.
  String? jumped;
  int? jumpedStart;
  await _try('epub-jump', () async {
    await _showChrome(tester);
    await _tapKey(tester, 'reader-header-contents');
    await _settle(tester, seconds: 1);
    final entry = find.descendant(
      of: find.byKey(const ValueKey('reader-contents-scroll')),
      matching: find.textContaining('CHAPTER L.', findRichText: true),
    );
    final scroll = find.descendant(
      of: find.byKey(const ValueKey('reader-contents-scroll')),
      matching: find.byType(Scrollable),
    );
    if (entry.evaluate().isEmpty) {
      await tester.scrollUntilVisible(
        entry,
        400,
        scrollable: scroll.first,
        maxScrolls: 80,
      );
    }
    await tester.ensureVisible(entry.first);
    await _settle(tester, ms: 600);
    await _shot(tester, 'epub-contents');
    await tester.tap(entry.first, warnIfMissed: false);
    await _settle(tester, seconds: 2);
    await _shot(tester, 'epub-contents-preview');
    await _goTo(tester);
    await _hideChrome(tester);
    final waited = await _waitNumbered(tester, seconds: 60);
    jumped = _folio(tester);
    jumpedStart = _last(_location(tester));
    _note('epub-jump-numbered-after-ms', waited);
    _note('epub-jump', _location(tester));
    await _shot(tester, 'epub-jumped-chapter-50');
    // One page on and back again, for good measure.
    await _tapAt(tester, .85, settleMs: 1000);
    _note('epub-jump-next', _location(tester));
    await _tapAt(tester, .15, settleMs: 1000);
    _note('epub-jump-back', _location(tester));
  });

  // 5b, continued: page from the cover to the chapter, animations off.
  await _try('epub-paging', () async {
    if (jumpedStart == null) throw StateError('no jump');
    await _setAnimations(tester, false);
    await _showChrome(tester);
    await _tapKey(tester, 'reader-header-contents');
    await _settle(tester, seconds: 1);
    final scroll = find.descendant(
      of: find.byKey(const ValueKey('reader-contents-scroll')),
      matching: find.byType(Scrollable),
    );
    final first = find.descendant(
      of: find.byKey(const ValueKey('reader-contents-scroll')),
      matching: find.textContaining('Chapter I.', findRichText: true),
    );
    if (first.evaluate().isEmpty) {
      await tester.scrollUntilVisible(
        first,
        -400,
        scrollable: scroll.first,
        maxScrolls: 80,
      );
    }
    await tester.ensureVisible(first.first);
    await _settle(tester, ms: 600);
    await tester.tap(first.first, warnIfMissed: false);
    await _settle(tester, seconds: 2);
    await _goTo(tester);
    await _hideChrome(tester);
    await _waitNumbered(tester, seconds: 60);
    _note('epub-paging-start', _location(tester));
    final start = DateTime.now();
    var taps = 0;
    final trail = <String>[];
    while (_last(_location(tester)) < jumpedStart! && taps < 900) {
      final before = _location(tester);
      await _tapAt(tester, .85, settleMs: 0);
      taps++;
      final until = DateTime.now().add(const Duration(seconds: 3));
      while (_location(tester) == before && DateTime.now().isBefore(until)) {
        await tester.pump(const Duration(milliseconds: 16));
      }
      await _settle(tester, ms: 60);
      if (taps % 25 == 0) trail.add(_location(tester));
    }
    await _waitNumbered(tester, seconds: 30);
    final paged = _folio(tester);
    _note('epub-paging-trail', trail.join(' ; '));
    _note(
      'epub-paging',
      '$taps taps in ${DateTime.now().difference(start).inSeconds}s -> '
          '${_location(tester)}',
    );
    _note('epub-paging-vs-jump', 'paged "$paged" jumped "$jumped" '
        'same ${paged == jumped} '
        'same-place ${_last(_location(tester)) == jumpedStart}');
    await _shot(tester, 'epub-paged-to-chapter-50');
    await _setAnimations(tester, true);
  });
  // 5c. Text size up and back keeps the number.
  await _try('epub-text-size', () async {
    final before = _location(tester);
    await _openAa(tester);
    await _tapKey(tester, 'reader-typography-font-increase');
    await _settle(tester, seconds: 2);
    await _dismissPanel(tester);
    await _hideChrome(tester);
    await _waitNumbered(tester, seconds: 60);
    final up = _location(tester);
    await _shot(tester, 'epub-text-larger');
    await _openAa(tester);
    await _tapKey(tester, 'reader-typography-font-decrease');
    await _settle(tester, seconds: 2);
    await _dismissPanel(tester);
    await _hideChrome(tester);
    await _waitNumbered(tester, seconds: 60);
    final back = _location(tester);
    for (final wait in [10, 30]) {
      await _settle(tester, seconds: wait == 10 ? 10 : 20);
      _note('epub-text-back-after-${wait}s', _location(tester));
    }
    await _shot(tester, 'epub-text-back');
    _note('epub-text-size', '$before -> larger $up -> back $back');
    _note('epub-text-size-same-folio', _folioOf(before) == _folioOf(back));
  });

}

/// Where the text sits: the stage, page, running head, text and folio.
void _pageArea(String label) {
  for (final k in [
    'reader-stage',
    'reader-page',
    'reader-page-tap-area',
    'reader-running-head',
    'reader-selection-surface',
    'reader-folio',
    'reader-header',
  ]) {
    _note('area $label $k', _keyRectsAny(k));
  }
}

String _keyRectsAny(String key) => [
  for (final e in find.byKey(ValueKey(key)).evaluate())
    if (_elementRect(e) case final r?) _rect(r),
].join(' ');

/// F2 exactly as in round 1: Contents -> CHAPTER L -> Go to; Aa larger; Aa
/// smaller; close Aa; ten seconds with the chrome hidden.
Future<void> _f2(WidgetTester tester) async {
  _marginTaps = true;
  await _try('f2', () async {
    if (_stage.evaluate().isNotEmpty) await _toLibrary(tester);
    await _open(tester, 'Pride');
    await _dismissHint(tester);
    await _hideChrome(tester);
    await _waitNumbered(tester);
    _note('f2-opened', _location(tester));
    await _showChrome(tester);
    await _tapKey(tester, 'reader-header-contents');
    await _settle(tester, seconds: 1);
    final entry = find.descendant(
      of: find.byKey(const ValueKey('reader-contents-scroll')),
      matching: find.textContaining('CHAPTER L.', findRichText: true),
    );
    await tester.ensureVisible(entry.first);
    await _settle(tester, ms: 600);
    await tester.tap(entry.first, warnIfMissed: false);
    await _settle(tester, seconds: 2);
    await _goTo(tester);
    await _hideChrome(tester);
    final waited = await _waitNumbered(tester, seconds: 60);
    _note('f2-jumped', '${_location(tester)} numbered after ${waited}ms');
    _pageArea('chrome-hidden');
    await _shot(tester, 'f2-1-jumped');
    await _showChrome(tester);
    final hold = await _holdChrome(tester);
    _pageArea('chrome-shown');
    await _shot(tester, 'f2-2-chrome-shown');
    await hold?.cancel();
    await _hideChrome(tester);
    _pageArea('chrome-hidden-again');
    _note('f2-before-text-size', _location(tester));

    _out('SHOSAI-MARK ---- text size change starts ----');
    await _openAa(tester);
    _note('f2-aa-open', _location(tester));
    _out('SHOSAI-MARK tap larger');
    await _tapKey(tester, 'reader-typography-font-increase');
    await _settle(tester, seconds: 2);
    final w1 = await _waitNumbered(tester, seconds: 60);
    _note('f2-larger', '${_location(tester)} numbered after ${w1}ms');
    await _shot(tester, 'f2-3-larger-panel-open');
    _out('SHOSAI-MARK tap smaller');
    await _tapKey(tester, 'reader-typography-font-decrease');
    await _settle(tester, seconds: 2);
    final w2 = await _waitNumbered(tester, seconds: 60);
    _note('f2-smaller-panel-open', '${_location(tester)} numbered after ${w2}ms');
    _out('SHOSAI-MARK close Aa');
    await _dismissPanel(tester);
    _note('f2-aa-closed', '${_location(tester)} chrome ${_chrome(tester)}');
    _out('SHOSAI-MARK hide chrome');
    await _hideChrome(tester);
    for (var i = 1; i <= 10; i++) {
      await _settle(tester, seconds: 1);
      _note('f2-hidden-${i}s', '${_location(tester)} chrome ${_chrome(tester)}');
    }
    _pageArea('after');
    await _shot(tester, 'f2-4-after-10s');
    _out('SHOSAI-MARK ---- end ----');
  });
}

/// The Aa panel's text size label, such as `90%`.
String _sizeLabel() => [
  for (final e in find.byType(Text).evaluate())
    if (e.widget case Text(:final data?) when RegExp(r'^\d+%$').hasMatch(data))
      data,
].join(',');

/// Aa + then Aa - at the current page: labels, folios and places.
Future<void> _sizeRoundTrip(WidgetTester tester, String tag) async {
  await _hideChrome(tester);
  await _waitNumbered(tester, seconds: 60);
  final start = _location(tester);
  _note('$tag-start', start);
  await _shot(tester, '$tag-1-start');
  await _openAa(tester);
  final l0 = _sizeLabel();
  await _tapKey(tester, 'reader-typography-font-increase');
  await _settle(tester, seconds: 2);
  await _waitNumbered(tester, seconds: 60);
  final l1 = _sizeLabel();
  final up = _location(tester);
  await _shot(tester, '$tag-2-larger');
  await _tapKey(tester, 'reader-typography-font-decrease');
  await _settle(tester, seconds: 2);
  await _waitNumbered(tester, seconds: 60);
  final l2 = _sizeLabel();
  final back = _location(tester);
  await _shot(tester, '$tag-3-back-panel-open');
  await _dismissPanel(tester);
  await _hideChrome(tester);
  await _settle(tester, seconds: 10);
  final after = _location(tester);
  await _shot(tester, '$tag-4-after-10s');
  _note('$tag-labels', '$l0 -> $l1 -> $l2');
  _note('$tag-places', '$start -> $up -> $back -> after 10s $after');
  _note('$tag-same', 'folios ${_folioOf(start) == _folioOf(after)} '
      'place ${start.split(' ').first == after.split(' ').first}');
}

Future<void> _contentsGo(WidgetTester tester, String label) async {
  await _showChrome(tester);
  await _tapKey(tester, 'reader-header-contents');
  await _settle(tester, seconds: 1);
  final entry = find.descendant(
    of: find.byKey(const ValueKey('reader-contents-scroll')),
    matching: find.textContaining(label, findRichText: true),
  );
  await tester.ensureVisible(entry.first);
  await _settle(tester, ms: 600);
  await tester.tap(entry.first, warnIfMissed: false);
  await _settle(tester, seconds: 2);
  await _goTo(tester);
  await _hideChrome(tester);
}

Future<void> _f2r3(WidgetTester tester) async {
  _marginTaps = true;
  await _try('f2-open', () async {
    if (_stage.evaluate().isNotEmpty) await _toLibrary(tester);
    await _open(tester, 'Pride');
    await _dismissHint(tester);
    await _hideChrome(tester);
    await _waitNumbered(tester);
  });
  await _try('f2-chapter-l', () async {
    await _contentsGo(tester, 'CHAPTER L.');
    await _sizeRoundTrip(tester, 'f2a');
  });
  await _try('f2-mid-book', () async {
    await _setAnimations(tester, false);
    await _contentsGo(tester, 'Chapter I.');
    await _waitNumbered(tester, seconds: 60);
    for (var i = 0; i < 120; i++) {
      final before = _location(tester);
      await _tapAt(tester, .85, settleMs: 0);
      final until = DateTime.now().add(const Duration(seconds: 3));
      while (_location(tester) == before && DateTime.now().isBefore(until)) {
        await tester.pump(const Duration(milliseconds: 16));
      }
      await _settle(tester, ms: 60);
    }
    _note('f2b-paged-to', _location(tester));
    await _setAnimations(tester, true);
    await _sizeRoundTrip(tester, 'f2b');
  });
}

final _stage = find.byKey(const ValueKey('reader-stage'));

Future<void> _try(String name, Future<void> Function() body) async {
  try {
    await body();
    _note('$name-ok', true);
  } catch (error, stack) {
    _note(
      '$name-FAILED',
      '$error ${stack.toString().split('\n').take(4).join(' | ')}',
    );
  }
}

void _note(String key, Object? value) => _out('QP $key: $value');

void _out(String line) {
  // ignore: avoid_print
  print(line);
  stdout.writeln(line);
}

Future<void> _settle(WidgetTester tester, {int seconds = 0, int ms = 0}) async {
  final end = DateTime.now().add(Duration(seconds: seconds, milliseconds: ms));
  while (DateTime.now().isBefore(end)) {
    await tester.pump(const Duration(milliseconds: 16));
  }
}

String _stats(List<int> v) {
  if (v.isEmpty) return 'none';
  final s = [...v]..sort();
  final mean = s.reduce((a, b) => a + b) / s.length;
  return 'n=${s.length} min ${s.first} median ${s[s.length ~/ 2]} '
      'mean ${mean.toStringAsFixed(1)} max ${s.last} all $v';
}

bool _chrome(WidgetTester tester) => find
    .byKey(const ValueKey('reader-header-contents'))
    .hitTestable()
    .evaluate()
    .isNotEmpty;

Finder get _curlFinder => find.byKey(const ValueKey('reader-page-curl'));

bool _curl() => _curlFinder.evaluate().isNotEmpty;

/// The curl painter's progress and whether its incoming page is captured.
(double?, bool) _curlState(WidgetTester tester) {
  if (!_curl()) return (null, false);
  try {
    final paint = tester.widget<CustomPaint>(_curlFinder.first);
    final dynamic painter = paint.painter;
    return (painter.progress as double, painter.incoming != null);
  } catch (_) {
    return (null, false);
  }
}

bool _spread(WidgetTester tester) =>
    find.byKey(const ValueKey('reader-spread-gutter')).evaluate().isNotEmpty ||
    find.byKey(const ValueKey('reader-folio')).evaluate().length > 1;

/// The places on screen: EPUB canonical starts or PDF units, low first.
String _location(WidgetTester tester) {
  final places = <int>[];
  for (final p in tester.widgetList<CustomPaint>(find.byType(CustomPaint))) {
    final painter = p.painter;
    if (painter is ReaderEpubPageContentPainter) {
      // The canonical start restarts in each spine unit: place = unit:start.
      places.add(painter.page.unit * 10000000 + painter.page.page.canonicalStart);
    } else if (painter is PagePainter && painter.image != null) {
      places.add(painter.currentUnit);
    }
  }
  places.sort();
  return '${places.toSet().join(',')} [${_folio(tester)}]';
}

String _folioOf(String location) {
  final i = location.indexOf('[');
  return i < 0 ? '' : location.substring(i);
}

int _last(String location) {
  final places = location.split(' ').first.split(',');
  return places.map((p) => int.tryParse(p) ?? -1).fold(-1, (a, b) => a > b ? a : b);
}

int _first(String location) =>
    int.tryParse(location.split(RegExp('[ ,]')).first) ?? -1;

String _folio(WidgetTester tester) => [
  for (final e in find.byKey(const ValueKey('reader-folio')).evaluate())
    if (e.widget is Text) (e.widget as Text).data ?? '',
].join(' | ');

/// Waits until every folio on screen has a number; returns the wait in ms.
Future<int> _waitNumbered(WidgetTester tester, {int seconds = 20}) async {
  final start = DateTime.now();
  final end = start.add(Duration(seconds: seconds));
  bool numbered() {
    final texts = [
      for (final e in find.byKey(const ValueKey('reader-folio')).evaluate())
        if (e.widget is Text) (e.widget as Text).data ?? '',
    ];
    return texts.isNotEmpty && texts.every((t) => RegExp(r'\d').hasMatch(t));
  }

  while (!numbered() && DateTime.now().isBefore(end)) {
    await tester.pump(const Duration(milliseconds: 50));
  }
  return DateTime.now().difference(start).inMilliseconds;
}

String _rects(WidgetTester tester, Finder finder) => [
  for (final e in finder.evaluate())
    if (e.renderObject case final RenderBox box? when box.hasSize && box.attached)
      _rect(box.localToGlobal(Offset.zero) & box.size),
].join(' ');

String _keyRects(WidgetTester tester, String key) =>
    _rects(tester, find.byKey(ValueKey(key)));

String _insets(EdgeInsets e) => 'L${e.left} T${e.top} R${e.right} B${e.bottom}';

String _rect(Rect r) =>
    '(${r.left.toStringAsFixed(1)},${r.top.toStringAsFixed(1)} '
    '${r.width.toStringAsFixed(1)}x${r.height.toStringAsFixed(1)})';

Rect _tapArea(WidgetTester tester) {
  final area = find.byKey(const ValueKey('reader-page-tap-area'));
  final rects = [
    for (final e in area.evaluate())
      if (_elementRect(e) case final r?) r,
  ];
  if (rects.isEmpty) return tester.getRect(_stage.first);
  return rects.reduce((a, b) => a.expandToInclude(b));
}

Rect? _elementRect(Element e) {
  final box = e.renderObject;
  if (box is! RenderBox || !box.hasSize || !box.attached) return null;
  return box.localToGlobal(Offset.zero) & box.size;
}

Map<String, String> _geometry(WidgetTester tester) {
  final context = _stage.evaluate().isNotEmpty
      ? _stage.evaluate().first
      : tester.element(find.byType(Scaffold).first);
  final query = MediaQuery.of(context);
  return {
    'padding': _insets(query.padding),
    'size': '${query.size.width}x${query.size.height}',
    'stage': _keyRects(tester, 'reader-stage'),
    'tap-area': _keyRects(tester, 'reader-page-tap-area'),
    'page': _keyRects(tester, 'reader-page'),
    'pdf-page': _keyRects(tester, 'reader-pdf-page-shadow'),
    'folio': _keyRects(tester, 'reader-folio'),
    'location': _location(tester),
    'header': _keyRects(tester, 'reader-header'),
    'scrim': _keyRects(tester, 'reader-panel-scrim'),
    'chrome': '${_chrome(tester)}',
  };
}

Future<void> _shot(WidgetTester tester, String name) async {
  final error = tester.takeException();
  if (error != null) {
    _note('EXCEPTION-before-$name', '$error'.split('\n').take(3).join(' '));
  }
  await _settle(tester, ms: 600);
  _step += 1;
  final label = '$_device-${_step.toString().padLeft(2, '0')}-$name';
  _geometry(tester).forEach((k, v) => _note('$label $k', v));
  await _capture(tester, 'shot-$label', ratio: 1);
  _out('SHOT $label');
  await _settle(tester, seconds: 4);
}

Future<TestGesture?> _holdChrome(WidgetTester tester) async {
  final title = find.byKey(const ValueKey('reader-header-title'));
  if (title.evaluate().isEmpty || !_chrome(tester)) return null;
  final gesture = await tester.startGesture(tester.getCenter(title.first));
  await tester.pump(const Duration(milliseconds: 16));
  return gesture;
}

Future<void> _hideChrome(WidgetTester tester) async {
  if (_chrome(tester)) await _tapAt(tester, .5);
  if (_chrome(tester)) await _settle(tester, seconds: 4);
}

Future<void> _showChrome(WidgetTester tester) async {
  if (!_chrome(tester)) await _tapAt(tester, .5, settleMs: 600);
  if (!_chrome(tester)) {
    // A margin tap can miss on a phone page; the middle of the page shows it.
    final area = _tapArea(tester);
    await tester.tapAt(area.center);
    await _settle(tester, ms: 600);
  }
  _note('chrome-shown', _chrome(tester));
}

Future<void> _dismissHint(WidgetTester tester) async {
  final hint = find.byKey(const ValueKey('reader-zones-hint'));
  if (hint.evaluate().isNotEmpty) await _tapAt(tester, .5);
}

Future<void> _tapAt(
  WidgetTester tester,
  double fraction, {
  int settleMs = 1000,
}) async {
  final rect = _tapArea(tester);
  await tester.tapAt(Offset(rect.left + rect.width * fraction, _tapY(tester, rect)));
  if (settleMs > 0) await _settle(tester, ms: settleMs);
}

double _tapY(WidgetTester tester, Rect area) {
  if (!_marginTaps) return area.center.dy;
  final surface = find.byKey(const ValueKey('reader-selection-surface'));
  final rects = [
    for (final e in surface.evaluate())
      if (_elementRect(e) case final r?) r,
  ];
  if (rects.isEmpty) return area.center.dy;
  final y = rects.map((r) => r.bottom).reduce((a, b) => a > b ? a : b) + 6;
  return y < area.bottom - 2 ? y : area.center.dy;
}

typedef _Timed = ({String summary, int? firstCurlMs, int? firstMoveMs});

/// A touch tap in the turn zone, then frame-by-frame: the curl, its progress,
/// whether the incoming page is captured, and the folio widget's text. With
/// [capture], every frame is also captured in-app (no encoding until after).
Future<_Timed> _timedTap(
  WidgetTester tester,
  double f,
  String name, {
  required bool capture,
}) async {
  final rect = _tapArea(tester);
  final at = Offset(rect.left + rect.width * f, _tapY(tester, rect));
  final before = _location(tester);
  final folioBefore = _folio(tester);
  final images = <(int, ui.Image)>[];
  if (capture) images.add((-1, _grab()!));
  final clock = Stopwatch()..start();
  final gesture = await tester.startGesture(at);
  await gesture.up();
  final upMs = clock.elapsedMilliseconds;
  final rows = <String>[];
  int? firstCurl, firstMove, folioChange, lastCurl;
  while (clock.elapsedMilliseconds < 1500) {
    await tester.pump();
    final ms = clock.elapsedMilliseconds;
    final (progress, incoming) = _curlState(tester);
    final curl = _curl();
    if (curl) {
      firstCurl ??= ms;
      lastCurl = ms;
      if ((progress ?? 0) > 0) firstMove ??= ms;
    }
    final folio = _folio(tester);
    if (folio != folioBefore) folioChange ??= ms;
    rows.add(
      '$ms:${curl ? 'C' : '-'}${progress?.toStringAsFixed(2) ?? ''}'
      '${incoming ? 'i' : ''}${folio != folioBefore ? '*' : ''}',
    );
    if (capture && images.length < 90) {
      final image = _grab();
      if (image != null) images.add((ms, image));
    }
    if (!curl && lastCurl != null && ms - lastCurl > 200) break;
  }
  for (final (i, (ms, image)) in images.indexed) {
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    image.dispose();
    File(
      '${_frames.path}/$_device-$name-${i.toString().padLeft(3, '0')}-$ms.png',
    ).writeAsBytesSync(bytes!.buffer.asUint8List());
  }
  final summary =
      '$before -> ${_location(tester)}; up ${upMs}ms; curl first ${firstCurl}ms '
      'leaf moves ${firstMove}ms last ${lastCurl}ms; folio widget changed '
      '${folioChange}ms; frames ${rows.join(' ')}';
  return (summary: summary, firstCurlMs: firstCurl, firstMoveMs: firstMove);
}

ui.Image? _grab({double ratio = 1}) {
  try {
    final boundary =
        _captureKey.currentContext!.findRenderObject()! as RenderRepaintBoundary;
    return boundary.toImageSync(pixelRatio: ratio);
  } catch (error) {
    _note('grab-failed', error);
    return null;
  }
}

Future<void> _capture(WidgetTester tester, String name, {double ratio = 1}) async {
  try {
    final boundary =
        _captureKey.currentContext!.findRenderObject()! as RenderRepaintBoundary;
    final image = await boundary.toImage(pixelRatio: ratio);
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    image.dispose();
    File('${_frames.path}/$_device-$name.png')
        .writeAsBytesSync(bytes!.buffer.asUint8List());
  } catch (error) {
    _note('capture-$name-failed', error);
  }
}

/// A slow forward drag held at about a third of the leaf, a device
/// screenshot while held, then released back.
Future<void> _heldDrag(WidgetTester tester, String tag) async {
  final area = _tapArea(tester);
  final spread = _spread(tester);
  final y = area.center.dy;
  Offset at(double f) => Offset(area.left + area.width * f, y);
  final from = at(.95);
  final to = at(spread ? .75 : .70);
  final images = <(int, ui.Image)>[(-1, _grab()!)];
  final clock = Stopwatch()..start();
  final gesture = await tester.startGesture(from);
  const steps = 24;
  for (var i = 1; i <= steps; i++) {
    await gesture.moveTo(Offset.lerp(from, to, i / steps)!);
    await tester.pump(const Duration(milliseconds: 25));
    final image = _grab();
    if (image != null) images.add((clock.elapsedMilliseconds, image));
  }
  await _settle(tester, ms: 400);
  _note('$tag-held-curl', _curl());
  _geometry(tester).forEach((k, v) => _note('$tag-held $k', v));
  _step += 1;
  final label = '$_device-${_step.toString().padLeft(2, '0')}-$tag-curl-held';
  _out('SHOT $label');
  await _settle(tester, seconds: 4);
  final heldImage = _grab();
  if (heldImage != null) images.add((clock.elapsedMilliseconds, heldImage));
  await gesture.up();
  for (var i = 0; i < 30; i++) {
    await tester.pump();
    final image = _grab();
    if (image != null) images.add((clock.elapsedMilliseconds, image));
    if (!_curl()) break;
  }
  await _settle(tester, seconds: 1);
  _note('$tag-held-release-curl', _curl());
  for (final (i, (ms, image)) in images.indexed) {
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    image.dispose();
    File(
      '${_frames.path}/$_device-$tag-held-${i.toString().padLeft(3, '0')}-$ms.png',
    ).writeAsBytesSync(bytes!.buffer.asUint8List());
  }
}

/// A Contents choice opens a preview of the chapter; Go to jumps there.

  final go = find.byKey(const ValueKey('reader-reference-go-to'));
  _note('go-to-shown', go.evaluate().isNotEmpty);
  if (go.evaluate().isEmpty) return;
  await tester.tap(go.first, warnIfMissed: false);
  await _settle(tester, seconds: 2);
}

Future<void> _openAa(WidgetTester tester) async {
  await _showChrome(tester);
  await _tapKey(tester, 'reader-header-typography');
  await _settle(tester, seconds: 1);
}

/// The Aa panel open: where its scrim starts against the status bar and
/// toolbar, with a device screenshot.
Future<void> _aaPanel(WidgetTester tester, String tag) async {
  await _hideChrome(tester);
  await _openAa(tester);
  final hold = await _holdChrome(tester);
  _note('$tag-aa-scrim', _keyRects(tester, 'reader-panel-scrim'));
  _note('$tag-aa-header', _keyRects(tester, 'reader-header'));
  await _shot(tester, '$tag-aa-panel');
  await hold?.cancel();
  await _dismissPanel(tester);
  await _hideChrome(tester);
}

Future<void> _setAnimations(WidgetTester tester, bool on) async {
  await _openAa(tester);
  final toggle = find.byKey(const ValueKey('reader-typography-animations'));
  await tester.ensureVisible(toggle.first);
  await _settle(tester, ms: 600);
  final dynamic widget = toggle.evaluate().first.widget;
  bool? value;
  try {
    value = widget.value as bool?;
  } catch (_) {}
  if (value != on) {
    await tester.tap(toggle.first, warnIfMissed: false);
    await _settle(tester, ms: 600);
  }
  _note('animations', '$value -> $on');
  await _dismissPanel(tester);
  await _hideChrome(tester);
}

String _timingSummary(List<ui.FrameTiming> frames, WidgetTester tester) {
  if (frames.isEmpty) return 'no frames';
  double ms(Duration d) => d.inMicroseconds / 1000;
  final build = frames.map((f) => ms(f.buildDuration)).toList()..sort();
  final raster = frames.map((f) => ms(f.rasterDuration)).toList()..sort();
  final hz = tester.view.display.refreshRate;
  final budget = 1000 / (hz > 0 ? hz : 60);
  String p(List<double> v, double q) =>
      v[((v.length - 1) * q).round()].toStringAsFixed(2);
  return 'frames=${frames.length} refresh=${hz}Hz budget=${budget.toStringAsFixed(1)}ms '
      'build p50/p95/max=${p(build, .5)}/${p(build, .95)}/${build.last.toStringAsFixed(2)} '
      'raster p50/p95/max=${p(raster, .5)}/${p(raster, .95)}/${raster.last.toStringAsFixed(2)} '
      'over-budget=${frames.where((f) => ms(f.buildDuration) > budget || ms(f.rasterDuration) > budget).length}';
}

Future<void> _tapKey(WidgetTester tester, String key) async {
  final finder = find.byKey(ValueKey(key));
  if (finder.evaluate().isEmpty) throw StateError('no $key on screen');
  await tester.ensureVisible(finder.first);
  await tester.tap(finder.first, warnIfMissed: false);
  await _settle(tester, ms: 500);
}

Future<void> _dismissPanel(WidgetTester tester) async {
  final close = find.byKey(const ValueKey('reader-typography-close'));
  final scrim = find.byKey(const ValueKey('reader-panel-scrim'));
  if (scrim.evaluate().isNotEmpty) {
    await tester.tapAt(
      tester.getRect(scrim.first).topLeft + const Offset(24, 16),
    );
  } else if (close.evaluate().isNotEmpty) {
    await tester.tap(close.first, warnIfMissed: false);
  } else {
    await _tapKey(tester, 'reader-header-typography');
  }
  await _settle(tester, seconds: 1);
}

Future<void> _open(WidgetTester tester, String title) async {
  final finder = find.textContaining(title, findRichText: true);
  final scrollable = find.byType(Scrollable);
  if (finder.evaluate().isEmpty && scrollable.evaluate().isNotEmpty) {
    await tester.scrollUntilVisible(
      finder,
      300,
      scrollable: scrollable.first,
      maxScrolls: 30,
    );
  }
  if (finder.evaluate().isEmpty) throw StateError('no "$title" on screen');
  await tester.ensureVisible(finder.last);
  await _settle(tester, seconds: 1);
  await tester.tap(finder.last, warnIfMissed: false);
  await _settle(tester, seconds: 5);
}

Future<void> _toLibrary(WidgetTester tester) async {
  await _showChrome(tester);
  if (find.byKey(const ValueKey('reader-header-back')).evaluate().isNotEmpty) {
    await _tapKey(tester, 'reader-header-back');
  } else {
    var library = find.byTooltip(_localeTag == 'ja' ? 'ライブラリ' : 'Library');
    if (library.evaluate().isEmpty) {
      library = find.text(_localeTag == 'ja' ? 'ライブラリ' : 'Library');
    }
    if (library.evaluate().isEmpty) throw StateError('no way to the library');
    await tester.tap(library.first, warnIfMissed: false);
  }
  await _settle(tester, seconds: 3);
}
